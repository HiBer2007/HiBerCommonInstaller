# HiBerCommonInstaller - generic installer framework (core + flow + ext + shells)
cmake_minimum_required(VERSION 3.20)
project(HiBerCommonInstaller VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# MSVC: make source AND execution charset UTF-8 so \uXXXX escapes in string
# literals (i18n table) compile to UTF-8 bytes, matching fromUtf8 decoding
# in the shells. Without /utf-8 the escapes follow the ANSI codepage (GBK)
# and every translated string is mangled.
if(MSVC)
    add_compile_options(/utf-8)
endif()
# AUTOMOC is enabled per-target: the core chain (zero Qt) must not see it;
# the GUI shell enables it locally in gui/CMakeLists.txt (M4).

# ------------------------------------------------------------------
# 版本定义（单点）：宿主可在 add_subdirectory 前覆盖 HCI_VERSION
# （如 NSUM 特化 = NSUM_INSTALLER_VERSION）。产物名与版本资源全部依赖它。
# ------------------------------------------------------------------
if(NOT DEFINED HCI_VERSION)
    set(HCI_VERSION "${PROJECT_VERSION}")
endif()

option(HCI_BUILD_CLI   "Build hci_cli shell (console)"            ON)
option(HCI_BUILD_GUI   "Build hci_gui_shell (Qt + HiBerGUI)"       OFF)
option(HCI_BUILD_TUI   "Build hci_tui_shell (C++ ANSI)"           OFF)
option(HCI_BUILD_EXAMPLES "Build examples (demo product)"         ON)

# Default: the bundled installer.ico; hosts may override via

    CACHE FILEPATH "Icon (.ico) embedded into the shell executables")

# ------------------------------------------------------------------
# Version resource helper (self-contained; no icon by default)
# ------------------------------------------------------------------
function(hci_add_version_info target desc product)
    if(NOT TARGET ${target})
        message(FATAL_ERROR "hci_add_version_info: unknown target ${target}")
    endif()
    # 版本号依赖 HCI_VERSION（宿主可覆盖）；拆分四段写入 VERSIONINFO
    set(_hci_v0 0)
    set(_hci_v1 0)
    set(_hci_v2 0)
    set(_hci_v3 0)
    string(REPLACE "." ";" _hci_ver_parts "${HCI_VERSION}")
    list(LENGTH _hci_ver_parts _hci_ver_n)
    if(_hci_ver_n GREATER 0)
        list(GET _hci_ver_parts 0 _hci_v0)
    endif()
    if(_hci_ver_n GREATER 1)
        list(GET _hci_ver_parts 1 _hci_v1)
    endif()
    if(_hci_ver_n GREATER 2)
        list(GET _hci_ver_parts 2 _hci_v2)
    endif()
    if(_hci_ver_n GREATER 3)
        list(GET _hci_ver_parts 3 _hci_v3)
    endif()
    get_target_property(_hci_out ${target} OUTPUT_NAME)
    if(NOT _hci_out)
        set(_hci_out "${target}")
    endif()

    set(_hci_vr "${CMAKE_CURRENT_BINARY_DIR}/version_${target}.rc")
    set(_hci_content "// AUTO-GENERATED by hci_add_version_info\n")
    string(APPEND _hci_content "1 VERSIONINFO\n")
    string(APPEND _hci_content " FILEVERSION ${_hci_v0},${_hci_v1},${_hci_v2},${_hci_v3}\n")
    string(APPEND _hci_content " PRODUCTVERSION ${_hci_v0},${_hci_v1},${_hci_v2},${_hci_v3}\n")
    string(APPEND _hci_content " FILEFLAGSMASK 0x3fL\n")
    string(APPEND _hci_content " FILEFLAGS 0x0L\n")
    string(APPEND _hci_content " FILEOS 0x40004L\n")
    string(APPEND _hci_content " FILETYPE 0x1L\n")
    string(APPEND _hci_content " FILESUBTYPE 0x0L\n")
    string(APPEND _hci_content "BEGIN\n")
    string(APPEND _hci_content "    BLOCK \"StringFileInfo\"\n")
    string(APPEND _hci_content "    BEGIN\n")
    string(APPEND _hci_content "        BLOCK \"040904b0\"\n")
    string(APPEND _hci_content "        BEGIN\n")
    string(APPEND _hci_content "            VALUE \"CompanyName\", \"HiBer2007\"\n")
    string(APPEND _hci_content "            VALUE \"FileDescription\", \"${desc}\"\n")
    string(APPEND _hci_content "            VALUE \"FileVersion\", \"${HCI_VERSION}\"\n")
    string(APPEND _hci_content "            VALUE \"InternalName\", \"${target}\"\n")
    string(APPEND _hci_content "            VALUE \"LegalCopyright\", \"Copyright (C) 2026 HiBer2007\"\n")
    string(APPEND _hci_content "            VALUE \"OriginalFilename\", \"${_hci_out}.exe\"\n")
    string(APPEND _hci_content "            VALUE \"ProductName\", \"${product}\"\n")
    string(APPEND _hci_content "            VALUE \"ProductVersion\", \"${HCI_VERSION}\"\n")
    string(APPEND _hci_content "        END\n")
    string(APPEND _hci_content "    END\n")
    string(APPEND _hci_content "    BLOCK \"VarFileInfo\"\n")
    string(APPEND _hci_content "    BEGIN\n")
    string(APPEND _hci_content "        VALUE \"Translation\", 0x0409, 1200\n")
    string(APPEND _hci_content "    END\n")
    string(APPEND _hci_content "END\n")
    # Optional embedded icon (HCI_ICON_PATH; rc paths use forward slashes).
    if(EXISTS "${HCI_ICON_PATH}")
        string(REPLACE "\\" "/" _hci_icon "${HCI_ICON_PATH}")
        string(APPEND _hci_content "\nIDI_ICON1 ICON \"${_hci_icon}\"\n")
    endif()
    set(_hci_utf8bom "")
    string(ASCII 239 187 191 _hci_utf8bom)
    file(WRITE "${_hci_vr}" "${_hci_utf8bom}${_hci_content}")
    target_sources(${target} PRIVATE "${_hci_vr}")
endfunction()
